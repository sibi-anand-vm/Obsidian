## Add User To MOCK API

Get below free API from MOCK API website.
```
https://67a177985bcfff4fabe27929.mockapi.io/users
```

Create an folder named **mock-users** in app folder with page.tsx.

Copy paste the following code to fetch data from API and add user to it.
```
type User={
id:number,
name:string
};

export default async function Mockusers() {

let res=await fetch('https://67a177985bcfff4fabe27929.mockapi.io/users');
let users:User[]=await res.json();

async function adduser(formData:FormData){
"use server"
let name=formData.get("name");
let r=await fetch("https://67a177985bcfff4fabe27929.mockapi.io/users",{
method:"POST",
headers:{"Content-type":"application/json"},
body:JSON.stringify({name})
})
}

return(
<>
<form action={adduser}>
<input type="name" name="name"></input>
<button type="submit">Add User</button>
</form>

{users.map((user)=>(
<div key={user.id}>
<h1>{user.name}</h1>
</div>
))}
</>
)}
```
