2.Copy paste the following code for checking Data Fetching from external API :<font color="#4bacc6">https://jsonplaceholder.typicode.com/users</font>
```
"use client"
import { useState,useEffect } from "react"

export default function UsersFetch(){
type User = {
id: number;
name: string;
email: string;
};
const [users,setusers]=useState<User[]>([]);
useEffect(()=>{
async function fetchusers(){
try{
let response=await fetch('https://jsonplaceholder.typicode.com/users');
if(!response.ok) {console.log("Cannot able to fetch users");
return;
}

let data=await response.json();
setusers(data);

console.log("Data downloaded")
}
catch(err){
console.log("Error occur");
}
}
fetchusers();
},[])

return
<>
{users.map((user)=>(
<div key={user.id}>
<h1>{user.name}</h1>
</div>
))}
</>
);
}
```
