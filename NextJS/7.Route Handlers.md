1.Create users folder in app folder with creating route.ts & <font color="#e5e0ec">usersarray.js</font> in it.
```
import users from './usersarray'
export function GET(){
return Response.json(users);
}

export async function POST(request:Request){
const user=await request.json();
let newuser={
id:users.length+1,
name:user.name,
}

users.push(newuser);
return new Response(JSON.stringify(newuser),{
"headers":{"Content-type":"application/json"}
,status:200,
}
);
}
```

2.Create sub folder [id] with route.ts in users folder for fetching individual user.
```
import users from '../usersarray'
export async function GET(request:Request,{params}:{params:{id:string}}){
const {id}=await params;
const user=users.find((user)=>user.id===parseInt(id))
return Response.json(user);
}
```
